from math import isclose

def memory_bytes(n, m):
    if n < 2 or m < 0 or m > n * (n - 1) // 2:
        raise ValueError("undirected simple graph contract")
    dense = 8 * n * n
    csr = 8 * (n + 1) + 16 * (2 * m)
    return dense, csr

def star_center_probs(leaves):
    if leaves < 1:
        raise ValueError("nonempty star required")
    uniform = 1 / (leaves + 1)
    random_walk_stationary = leaves / (2 * leaves)
    return uniform, random_walk_stationary

dense, csr = memory_bytes(1_000_000, 5_000_000)
assert dense == 8_000_000_000_000
assert csr == 168_000_008
uniform, walk = star_center_probs(5)
assert isclose(uniform, 1 / 6)
assert isclose(walk, .5)
print("theoretical bytes:", dense, csr)
