from math import isclose

weights = {("A", "B"): .8, ("B", "C"): .6, ("A", "C"): .2}

def threshold_graph(tau):
    edges = [(a, b) for (a, b), w in weights.items()
             if w + 1e-12 >= tau]
    adj = {v: set() for v in "ABC"}
    for a, b in edges:
        adj[a].add(b)
        adj[b].add(a)
    visited = set()
    components = 0
    for node in adj:
        if node in visited:
            continue
        components += 1
        todo = [node]
        while todo:
            v = todo.pop()
            if v in visited:
                continue
            visited.add(v)
            todo.extend(adj[v] - visited)
    return len(edges), components

assert threshold_graph(.5) == (2, 1)
assert threshold_graph(.7) == (1, 2)
exposure_rate = (2 + 32) / (20 + 80)
control_rate = (8 + 8) / (80 + 20)
assert isclose(exposure_rate - control_rate, .18)
assert isclose(2/20 - 8/80, 0)
assert isclose(32/80 - 8/20, 0)
assert isclose(abs(3/4 - 1/4), .5)
print("brain edges/components:", threshold_graph(.7),
      "observational risk diff:", exposure_rate-control_rate)
