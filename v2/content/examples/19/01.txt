import networkx as nx

actors = list("ABCD")
social = [("A","B"),("B","C")]
collab = [("A","C"),("C","D")]
supra = nx.Graph()
for layer in ("social","collab"):
    supra.add_nodes_from((v,layer) for v in actors)
for u,v in social:
    supra.add_edge((u,"social"),(v,"social"))
for u,v in collab:
    supra.add_edge((u,"collab"),(v,"collab"))
for v in actors:
    supra.add_edge((v,"social"),(v,"collab"))

start, goal = ("B","social"), ("D","collab")
assert (supra.number_of_nodes(),supra.number_of_edges()) == (8,8)
assert nx.shortest_path_length(supra,start,goal) == 3

blocked = supra.copy()
blocked.remove_edges_from([((v,"social"),(v,"collab")) for v in actors])
assert not nx.has_path(blocked,start,goal)

flat = nx.Graph(social+collab)
assert nx.shortest_path_length(flat,"B","D") == 2
print("supra path:", nx.shortest_path(supra,start,goal))
