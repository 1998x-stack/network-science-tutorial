import networkx as nx
import matplotlib.pyplot as plt

# ER随机图 —— 1000节点，连接概率0.01
G_er = nx.erdos_renyi_graph(1000, 0.01)
print(f"ER: N={G_er.number_of_nodes()}, E={G_er.number_of_edges()}")
print(f"  平均度: {2 * G_er.number_of_edges() / G_er.number_of_nodes():.1f}")

# WS小世界网络 —— 1000节点，每个节点与20个邻居相连，重连概率0.1
G_ws = nx.watts_strogatz_graph(1000, 20, 0.1)
print(f"WS: 平均聚类={nx.average_clustering(G_ws):.3f}, "
      f"平均路径={nx.average_shortest_path_length(G_ws):.3f}")

# BA无标度网络 —— 1000节点，每次新增节点连3条边
G_ba = nx.barabasi_albert_graph(1000, 3)
print(f"BA: N={G_ba.number_of_nodes()}, E={G_ba.number_of_edges()}")

# 验证幂律度分布
degree_seq = [d for n, d in G_ba.degree()]
max_deg = max(degree_seq)
print(f"  BA最大度: {max_deg}, 最小度: {min(degree_seq)}")
