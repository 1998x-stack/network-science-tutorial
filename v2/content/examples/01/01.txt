import networkx as nx

G = nx.Graph()
G.add_edges_from([
    ("Lin", "Mei"),
    ("Lin", "Jun"),
    ("Mei", "Jun"),
    ("Jun", "Kai"),
])

print("节点数:", G.number_of_nodes())
print("边数:", G.number_of_edges())
print("各节点度数:", dict(G.degree()))

# 无向图的握手定理：所有顶点度数之和等于边数的两倍。
assert sum(dict(G.degree()).values()) == 2 * G.number_of_edges()
assert nx.is_connected(G)
